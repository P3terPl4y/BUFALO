(function () {
    const el = document.getElementById('route-map');
    if (!el) return;

    const pickupLat  = parseFloat(el.dataset.pickupLat);
    const pickupLng  = parseFloat(el.dataset.pickupLng);
    const dropoffLat = parseFloat(el.dataset.dropoffLat);
    const dropoffLng = parseFloat(el.dataset.dropoffLng);

    if ([pickupLat, pickupLng, dropoffLat, dropoffLng].some(v => isNaN(v) || v === 0)) {
        el.outerHTML = '<div class="tf-map-empty"><i class="bi bi-geo-alt"></i>' +
                       '<h4>Coordenadas inválidas</h4></div>';
        return;
    }

    const pickup  = [pickupLat, pickupLng];
    const dropoff = [dropoffLat, dropoffLng];

    const map = L.map(el, {
        center: [21.5218, -77.7812],
        zoom: 7,
        scrollWheelZoom: true,
        zoomControl: true,
    });

    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
        attribution: '&copy; OpenStreetMap',
        maxZoom: 19,
    }).addTo(map);

    function makeIcon(kind) {
        const symbol = kind === 'pickup' ? 'box-seam-fill' : 'flag-fill';
        return L.divIcon({
            className: '',
            html: `<div class="tf-map-pin tf-map-pin--${kind}"><i class="bi bi-${symbol}"></i></div>`,
            iconSize: [34, 34],
            iconAnchor: [17, 34],
            popupAnchor: [0, -34],
        });
    }

    L.marker(pickup, { icon: makeIcon('pickup') })
        .addTo(map)
        .bindPopup(`<strong>Recogida</strong><small>${escapeHtml(el.dataset.pickupCity || '')}</small>`);

    L.marker(dropoff, { icon: makeIcon('dropoff') })
        .addTo(map)
        .bindPopup(`<strong>Entrega</strong><small>${escapeHtml(el.dataset.dropoffCity || '')}</small>`);

    if (typeof L.Geodesic !== 'undefined') {
        new L.Geodesic([pickup, dropoff], {
            className: 'tf-route-line',
            color: '#D5001C',
            weight: 4,
            opacity: 0.85,
            steps: 60,
        }).addTo(map);
    } else {
        L.polyline([pickup, dropoff], {
            color: '#D5001C',
            weight: 4,
            opacity: 0.85,
            dashArray: '10, 6',
        }).addTo(map);
    }

    const bounds = L.latLngBounds([pickup, dropoff]);
    map.fitBounds(bounds, { padding: [50, 50], maxZoom: 12 });

    document.querySelector('[data-map-recenter]')?.addEventListener('click', () => {
        map.fitBounds(bounds, { padding: [50, 50], maxZoom: 12 });
    });

    setTimeout(() => map.invalidateSize(), 250);
    window.addEventListener('resize', () => map.invalidateSize());

    function escapeHtml(s) {
        return String(s)
            .replace(/&/g, '&amp;')
            .replace(/</g, '&lt;')
            .replace(/>/g, '&gt;')
            .replace(/"/g, '&quot;');
    }
})();
</script>

<script>
    (function () {
        const form = document.getElementById('loadEditForm');
        if (!form) return;

        /* ─────── Preview en vivo ─────── */
        const preview = {
            title: form.querySelector('[data-preview="title"]'),
            pickup: form.querySelector('[data-preview="pickup"]'),
            dropoff: form.querySelector('[data-preview="dropoff"]'),
            weight: form.querySelector('[data-preview="weight"]'),
            rate: form.querySelector('[data-preview="rate"]')
        };

        const statusDot = form.querySelector('[data-status-dot]');
        const statusLabel = form.querySelector('[data-status-label]');

        const statusMeta = {
            open: { color: 'var(--tf-success)', label: 'Abierta' },
            assigned: { color: 'var(--tf-warning)', label: 'Asignada' },
            in_transit: { color: 'var(--tf-info)', label: 'En tránsito' },
            delivered: { color: 'var(--tf-red-500)', label: 'Entregada' },
            cancelled: { color: 'var(--tf-gray-500)', label: 'Cancelada' }
        };

        // Devuelve la ciudad de la opción seleccionada (data-city) o el nombre
        function selectedCity(id) {
            const el = document.getElementById(id);
            if (!el || !el.value) return '—';
            const opt = el.options[el.selectedIndex];
            return opt?.dataset.city || opt?.text?.split('—')[0].trim() || '—';
        }

        function updatePreview() {
            preview.title.textContent = document.getElementById('title').value.trim() || 'Sin título';
            preview.pickup.textContent = selectedCity('pickup_destination_id');
            preview.dropoff.textContent = selectedCity('dropoff_destination_id');

            const w = document.getElementById('weight').value;
            preview.weight.textContent = w ? `${w} kg` : '— kg';

            const r = document.getElementById('rate').value;
            const c = document.getElementById('rate_currency').value;
            preview.rate.textContent = r ? `$${r} ${c}` : `$— ${c}`;

            const status = form.querySelector('input[name="status"]:checked')?.value;
            if (status && statusMeta[status]) {
                statusDot.style.background = statusMeta[status].color;
                statusLabel.textContent = statusMeta[status].label;
            }
        }

        form.addEventListener('input', updatePreview);
        form.addEventListener('change', updatePreview);
        updatePreview();

        /* ─────── Estado del selector de status ─────── */
        form.querySelectorAll('.tf-status-option').forEach(opt => {
            const radio = opt.querySelector('input[type="radio"]');
            const sync = () => opt.classList.toggle('is-selected', radio.checked);
            radio.addEventListener('change', () => {
                form.querySelectorAll('.tf-status-option').forEach(o => o.classList.remove('is-selected'));
                sync();
            });
            sync();
        });

        /* ─────── Progreso dinámico ─────── */
        const blocks = ['status', 'identity', 'route', 'cargo', 'extras'];
        const stepEls = {};
        blocks.forEach(id => {
            stepEls[id] = document.querySelector(`.tf-progress__step[data-step="${id}"]`);
        });

        function updateProgress() {
            blocks.forEach(id => {
                const block = document.querySelector(`[data-block="${id}"]`);
                if (!block) return;
                const inputs = block.querySelectorAll('input:not([type="radio"]), textarea, select');
                const total = inputs.length;
                const filled = Array.from(inputs).filter(el => el.value.trim() !== '').length;
                const complete = total === 0 || (filled / total) >= 0.7;
                stepEls[id]?.classList.toggle('is-complete', complete);
            });

            const firstIncomplete = blocks.find(id => !stepEls[id]?.classList.contains('is-complete'));
            blocks.forEach(id => stepEls[id]?.classList.remove('is-current'));
            if (firstIncomplete) stepEls[firstIncomplete].classList.add('is-current');
        }

        form.addEventListener('input', updateProgress);
        updateProgress();
    })();
</script>

     JS — Progreso + Resumen en vivo + Auto-cálculo tarifa/km
     ═══════════════════════════════════════════════════════════════ -->
<script>
(function () {
    const form = document.getElementById('cargaCreateForm');
    if (!form) return;

    const summary = {
        route:     form.querySelector('[data-summary="route"]'),
        weight:    form.querySelector('[data-summary="weight"]'),
        equipment: form.querySelector('[data-summary="equipment"]'),
        distance:  form.querySelector('[data-summary="distance"]'),
        rate:      form.querySelector('[data-summary="rate"]'),
        rate_km:   form.querySelector('[data-summary="rate_km"]')
    };

    const blocks = ['identification', 'route', 'cargo', 'extras'];
    const stepEls = {};
    blocks.forEach(function (id) {
        stepEls[id] = document.querySelector('.tf-progress__step[data-step="' + id + '"]');
    });

    // Devuelve la ciudad de la opción seleccionada (data-city) o "—"
    function selectedCity(id) {
        const el = document.getElementById(id);
        if (!el || !el.value) return '—';
        const opt = el.options[el.selectedIndex];
        return (opt && opt.dataset && opt.dataset.city) || '—';
    }

    function fmtNum(v) {
        if (v === null || v === undefined || v === '') return null;
        const n = Number(v);
        return isNaN(n) ? null : n;
    }

    // Auto-calcular tarifa_por_km si el usuario no la escribió manualmente
    function autoTarifaKm() {
        const total = fmtNum(document.getElementById('tarifa_total').value);
        const dist  = fmtNum(document.getElementById('distancia_km').value);
        const tarifaKmEl = document.getElementById('tarifa_por_km');

        if (total !== null && dist && dist > 0 && tarifaKmEl.dataset.manual !== 'true') {
            tarifaKmEl.value = (total / dist).toFixed(4);
        }
    }

    function updateSummary() {
        const pickup  = selectedCity('origen_direccion_id');
        const dropoff = selectedCity('destino_direccion_id');
        const weight  = document.getElementById('peso_kg').value;
        const dist    = document.getElementById('distancia_km').value;
        const equipment = document.getElementById('tipo_equipo');
        const rate    = document.getElementById('tarifa_total').value;
        const currency = document.getElementById('moneda').value;
        const tarifaKm = document.getElementById('tarifa_por_km').value;

        summary.route.textContent = pickup + ' → ' + dropoff;

        summary.weight.textContent = weight ? weight + ' kg' : '—';
        summary.distance.textContent = dist ? dist + ' km' : '—';

        summary.equipment.textContent = (equipment && equipment.value)
            ? equipment.options[equipment.selectedIndex].text
            : '—';

        summary.rate.textContent = rate ? rate + ' ' + currency : '—';

        summary.rate_km.textContent = tarifaKm
            ? Number(tarifaKm).toFixed(4) + ' ' + currency + '/km'
            : '—';
    }

    function updateProgress() {
        blocks.forEach(function (id) {
            const block = document.getElementById('block-' + id);
            if (!block) return;
            const required = block.querySelectorAll('[required]');
            let filled = 0;
            required.forEach(function (el) { if (el.value.trim() !== '') filled++; });
            const complete = required.length === 0 || filled === required.length;

            const step = stepEls[id];
            if (!step) return;
            step.classList.toggle('is-complete', complete);
        });

        const firstIncomplete = blocks.find(function (id) {
            return stepEls[id] && !stepEls[id].classList.contains('is-complete');
        });
        blocks.forEach(function (id) { if (stepEls[id]) stepEls[id].classList.remove('is-current'); });
        if (firstIncomplete && stepEls[firstIncomplete]) {
            stepEls[firstIncomplete].classList.add('is-current');
        }
    }

    form.addEventListener('input', function () {
        autoTarifaKm();
        updateSummary();
        updateProgress();
    });
    form.addEventListener('change', function () {
        autoTarifaKm();
        updateSummary();
        updateProgress();
    });

    // Si el usuario edita tarifa_por_km manualmente, dejamos de auto-calcular
    const tarifaKmInput = document.getElementById('tarifa_por_km');
    tarifaKmInput.addEventListener('input', function () {
        this.dataset.manual = 'true';
    });
    // Init
    autoTarifaKm();
    updateSummary();
    updateProgress();
})();
</script>

<script>
(function () {
    const NOMINATIM_SEARCH  = 'https://nominatim.openstreetmap.org/search';
    const NOMINATIM_REVERSE = 'https://nominatim.openstreetmap.org/reverse';
    const COUNTRY_CODES     = 'cu';

    const CUBA_CENTER = [21.5218, -77.7812];
    const CUBA_ZOOM   = 7;

    const root = document.querySelector('[data-map-picker]');
    if (!root || typeof L === 'undefined') {
        console.warn('[map-picker] Leaflet no está disponible; el mapa no se inicializará.');
        window.__initRegisterMapPicker = function () {};
        window.__invalidateRegisterMap = function () {};
        return;
    }

    const searchInput  = root.querySelector('.tf-map-picker__search-input');
    const resultsBox   = root.querySelector('.tf-map-picker__results');
    const mapEl        = root.querySelector('[data-map]');
    const coordsEl     = root.querySelector('[data-coords]');
    const hintEl       = root.querySelector('[data-address-hint]');

    const latInput     = document.getElementById('latitude');
    const lngInput     = document.getElementById('longitude');
    const cityInput    = document.getElementById('city');
    const stateInput   = document.getElementById('state');
    const countryInput = document.getElementById('country');
    const postalInput  = document.getElementById('postal_code');
    const addressInput = document.getElementById('address');

    let map = null;
    let marker = null;

    function makeIcon() {
        return L.divIcon({
            className: '',
            html: '<div class="tf-map-pin tf-map-pin--pickup"><i class="bi bi-geo-alt-fill"></i></div>',
            iconSize: [32, 32],
            iconAnchor: [16, 32],
        });
    }

    function setMarker(lat, lng, opts) {
        opts = opts || {};
        if (!map) return;

        if (marker) {
            marker.setLatLng([lat, lng]);
        } else {
            marker = L.marker([lat, lng], { icon: makeIcon(), draggable: true }).addTo(map);
            marker.on('dragend', function () {
                const pos = marker.getLatLng();
                setMarker(pos.lat, pos.lng);
            });
        }

        if (opts.pan !== false) map.panTo([lat, lng], { animate: true });

        if (latInput) latInput.value = lat.toFixed(8);
        if (lngInput) lngInput.value = lng.toFixed(8);

        if (coordsEl) {
            coordsEl.dataset.empty = 'false';
            coordsEl.innerHTML = '<i class="bi bi-crosshair"></i> ' + lat.toFixed(5) + ', ' + lng.toFixed(5);
        }

        if (!opts.skipReverse) reverseGeocode(lat, lng);
    }

    let reverseTimer = null;
    function reverseGeocode(lat, lng) {
        clearTimeout(reverseTimer);
        if (hintEl) hintEl.textContent = 'Obteniendo dirección…';

        reverseTimer = setTimeout(function () {
            const url = NOMINATIM_REVERSE + '?format=json&lat=' + lat + '&lon=' + lng + '&zoom=18&addressdetails=1&accept-language=es';
            fetch(url, { headers: { Accept: 'application/json' } })
                .then(function (r) { return r.json(); })
                .then(function (data) {
                    const a = data.address || {};

                    if (addressInput && !addressInput.value.trim()) {
                        addressInput.value = a.road
                            ? (a.road + (a.house_number ? ' #' + a.house_number : ''))
                            : (a.neighbourhood || a.suburb || '');
                    }
                    if (cityInput && !cityInput.value.trim()) {
                        cityInput.value = a.city || a.town || a.village || a.municipality || '';
                    }
                    if (stateInput && !stateInput.value.trim()) {
                        stateInput.value = a.state || a.province || a.region || '';
                    }
                    if (countryInput && (!countryInput.value.trim() || countryInput.value === 'US')) {
                        countryInput.value = (a.country_code || '').toUpperCase() || 'CU';
                    }
                    if (postalInput && !postalInput.value.trim()) {
                        postalInput.value = a.postcode || '';
                    }

                    if (hintEl) hintEl.textContent = data.display_name || 'Ubicación seleccionada';
                })
                .catch(function () {
                    if (hintEl) hintEl.textContent = 'No se pudo obtener la dirección';
                });
        }, 400);
    }

    let searchTimer = null;
    if (searchInput) {
        searchInput.addEventListener('input', function () {
            clearTimeout(searchTimer);
            const q = searchInput.value.trim();
            if (q.length < 3) { resultsBox.hidden = true; return; }
            searchTimer = setTimeout(function () { runSearch(q); }, 450);
        });
    }

    function runSearch(q) {
        const url = NOMINATIM_SEARCH + '?format=json&q=' + encodeURIComponent(q)
                  + '&countrycodes=' + COUNTRY_CODES + '&limit=6&addressdetails=1&accept-language=es';
        fetch(url, { headers: { Accept: 'application/json' } })
            .then(function (r) { return r.json(); })
            .then(renderResults)
            .catch(function () {
                resultsBox.innerHTML = '<div class="tf-map-picker__empty">Error de búsqueda</div>';
                resultsBox.hidden = false;
            });
    }

    function renderResults(items) {
        if (!items || !items.length) {
            resultsBox.innerHTML = '<div class="tf-map-picker__empty">Sin resultados</div>';
            resultsBox.hidden = false;
            return;
        }
        resultsBox.innerHTML = items.map(function (it, idx) {
            return '<button type="button" class="tf-map-picker__result" data-idx="' + idx + '">'
                 + '<i class="bi bi-geo-alt-fill"></i>'
                 + '<span>' + escapeHtml(it.display_name)
                 + '<small>' + escapeHtml((it.type || '')) + '</small></span></button>';
        }).join('');
        resultsBox.hidden = false;
        resultsBox._items = items;

        resultsBox.querySelectorAll('.tf-map-picker__result').forEach(function (btn) {
            btn.addEventListener('click', function () {
                const it = resultsBox._items[Number(btn.dataset.idx)];
                if (!it) return;
                const lat = parseFloat(it.lat);
                const lng = parseFloat(it.lon);
                map.setView([lat, lng], 15);
                setMarker(lat, lng, { skipReverse: true, pan: false });
                if (hintEl) hintEl.textContent = it.display_name;

                const a = it.address || {};
                if (addressInput && !addressInput.value.trim() && (a.road || a.neighbourhood || a.suburb)) {
                    addressInput.value = a.road
                        ? a.road + (a.house_number ? ' #' + a.house_number : '')
                        : (a.neighbourhood || a.suburb);
                }
                if (cityInput && !cityInput.value.trim()) {
                    cityInput.value = a.city || a.town || a.village || a.municipality || '';
                }
                if (stateInput && !stateInput.value.trim()) {
                    stateInput.value = a.state || a.province || a.region || '';
                }
                if (countryInput && (!countryInput.value.trim() || countryInput.value === 'US')) {
                    countryInput.value = (a.country_code || '').toUpperCase() || 'CU';
                }
                if (postalInput && !postalInput.value.trim()) {
                    postalInput.value = a.postcode || '';
                }
                searchInput.value = it.display_name;
                resultsBox.hidden = true;
            });
        });
    }

    document.addEventListener('click', function (e) {
        if (mapRootSafe() && !root.contains(e.target)) resultsBox.hidden = true;
    });
    function mapRootSafe() { return !!root; }

    [latInput, lngInput].forEach(function (inp) {
        if (!inp) return;
        inp.addEventListener('change', function () {
            const lat = parseFloat(latInput.value);
            const lng = parseFloat(lngInput.value);
            if (!isNaN(lat) && !isNaN(lng) && lat !== 0 && lng !== 0 && map) {
                setMarker(lat, lng);
            }
        });
    });

    window.__initRegisterMapPicker = function () {
        if (map) return;
        map = L.map(mapEl, {
            center: CUBA_CENTER,
            zoom: CUBA_ZOOM,
            scrollWheelZoom: true,
            zoomControl: true,
        });
        L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
            attribution: '&copy; OpenStreetMap',
            maxZoom: 18,
        }).addTo(map);

        map.on('click', function (e) {
            setMarker(e.latlng.lat, e.latlng.lng);
        });

        const initLat = parseFloat(root.dataset.initialLat);
        const initLng = parseFloat(root.dataset.initialLng);
        if (!isNaN(initLat) && !isNaN(initLng) && initLat !== 0 && initLng !== 0) {
            map.setView([initLat, initLng], 14);
            setMarker(initLat, initLng, { skipReverse: true, pan: false });
        }
        setTimeout(function () { map.invalidateSize(); }, 150);
    };

    window.__invalidateRegisterMap = function () {
        if (map) setTimeout(function () { map.invalidateSize(); }, 100);
    };

    window.addEventListener('resize', function () { if (map) map.invalidateSize(); });

    function escapeHtml(s) {
        return String(s)
            .replace(/&/g, '&amp;').replace(/</g, '&lt;')
            .replace(/>/g, '&gt;').replace(/"/g, '&quot;');
    }
})();

(function () {

    // ─── Helpers ───
    function currentRole() {
        const r = form.querySelector('input[name="role"]:checked');
        return r ? r.value : '';
    }
    function currentMode() {
        const m = form.querySelector('input[name="empresa_mode"]:checked');
        return m ? m.value : 'none';
    }
    function tipoEmpresaPorRol(role) {
        if (role === 'chofer')     return 'carrier';
        if (role === 'publicador') return 'broker';
        return '';
    }
    function show(el, visible) {
        if (!el) return;
        el.style.display = visible ? '' : 'none';
    }
    function isFilled(el) {
        if (!el) return false;
        if (el.type === 'checkbox' || el.type === 'radio') return el.checked;
        return String(el.value || '').trim() !== '';
    }

    // ─── Empresas por tipo ───
    function renderEmpresaOptions(tipo) {
        const current = selectEmpresa.value;
        selectEmpresa.innerHTML = '<option value="">Selecciona una empresa…</option>';
        const list = empresasData[tipo] || [];
        list.forEach(function (e) {
            const opt = document.createElement('option');
            opt.value = e.id;
            opt.textContent = e.nombre;
            if (String(e.id) === current) opt.selected = true;
            selectEmpresa.appendChild(opt);
        });
        if (list.length === 0) {
            const opt = document.createElement('option');
            opt.disabled = true;
            opt.textContent = 'No hay empresas disponibles — puedes crear una nueva';
            selectEmpresa.appendChild(opt);
        }
    }

    function waitForVisible(el, cb, maxTries) {
        if (!el) return;
        maxTries = maxTries || 60;
        (function check(n) {
            if (el.offsetWidth > 0 && el.offsetHeight > 0) { cb(); return; }
            if (n <= 0) return;
            requestAnimationFrame(function () { check(n - 1); });
        })(maxTries);
    }

    // ─── Visibilidad + required dinámicos ───
    function updateSections() {
        const role = currentRole();
        const isChofer = (role === 'chofer');
        const isPublicador = (role === 'publicador');
        const mode = currentMode();
        const tipo = tipoEmpresaPorRol(role);

        show(blockLocation,     !!role);
        show(blockChofer,       isChofer);
        show(blockPublicador,   isPublicador);
        show(blockPreferences,  isChofer);
        show(blockAvailability, isChofer);
        show(blockEmpresa,      !!role);

        if (!role) { updateChecklist(); return; }

        if (hint) {
            hint.textContent = isChofer
                ? 'Como transportista puedes asociarte a una empresa de tipo Carrier, crear una nueva u omitir este paso.'
                : 'Como broker puedes asociarte a una empresa de tipo Broker, crear una nueva u omitir este paso.';
        }

        renderEmpresaOptions(tipo);

        // Modo empresa
        show(blockExisting, mode === 'existing');
        show(blockNew,      mode === 'new');
        selectEmpresa.required = (mode === 'existing');
        const nameLegalEl = document.getElementById('empresa_nombre_legal');
        if (nameLegalEl) nameLegalEl.required = (mode === 'new');

        // Required de perfil de rol
        const reqMap = {
            chofer_numero_licencia:              isChofer,
            chofer_tipo_licencia:                isChofer,
            chofer_anios_experiencia:            isChofer,
            publicador_numero_licencia_broker:   isPublicador,
            publicador_anios_experiencia:        isPublicador,
        };
        Object.keys(reqMap).forEach(function (name) {
            const el = form.querySelector('[name="' + name + '"]');
            if (el) el.required = reqMap[name];
        });

        // Mapa
        if (blockLocation.style.display !== 'none') {
            if (!mapInitialized && typeof window.__initRegisterMapPicker === 'function') {
                mapInitialized = true;
                waitForVisible(document.querySelector('.tf-map-picker__map'), function () {
                    window.__initRegisterMapPicker();
                });
            } else if (typeof window.__invalidateRegisterMap === 'function') {
                window.__invalidateRegisterMap();
            }
        }

        updateChecklist();
    }

    // ─── Checklist / Progreso ───
    const progressSteps = document.querySelectorAll('.tf-progress__step');

    function stepDone(stepName) {
        switch (stepName) {
            case 'personal': {
                const ok =
                    isFilled(document.getElementById('name')) &&
                    isFilled(document.getElementById('email')) &&
                    isFilled(document.getElementById('password'));
                return ok;
            }
            case 'role':
                return !!currentRole();

            case 'location': {
                const required = [
                    'city', 'state', 'country', 'radius',
                ];
                return required.every(function (id) {
                    return isFilled(document.getElementById(id));
                });
            }
            case 'perfil-rol': {
                const role = currentRole();
                if (role === 'chofer') {
                    return [
                        'chofer_numero_licencia',
                        'chofer_tipo_licencia',
                        'chofer_anios_experiencia',
                    ].every(function (id) {
                        return isFilled(document.getElementById(id));
                    });
                }
                if (role === 'publicador') {
                    return [
                        'publicador_numero_licencia_broker',
                        'publicador_anios_experiencia',
                    ].every(function (id) {
                        return isFilled(document.getElementById(id));
                    });
                }
                return false;
            }
            case 'empresa': {
                const role = currentRole();
                if (!role) return false;
                const mode = currentMode();
                if (mode === 'none') return true;
                if (mode === 'existing') return isFilled(selectEmpresa);
                if (mode === 'new') {
                    return isFilled(document.getElementById('empresa_nombre_legal'));
                }
                return false;
            }
        }
        return false;
    }

    function updateChecklist() {
        if (!progressSteps.length) return;

        let firstIncomplete = null;
        progressSteps.forEach(function (step) {
            const name = step.getAttribute('data-step');
            const done = stepDone(name);
            step.classList.toggle('is-complete', done);
            step.classList.remove('is-current');
            if (!done && !firstIncomplete) firstIncomplete = step;
        });

        if (firstIncomplete) {
            firstIncomplete.classList.add('is-current');
        } else if (progressSteps.length) {
            // Todo completo → marcamos el último como current
            progressSteps[progressSteps.length - 1].classList.add('is-current');
        }
    }

    // ─── Listeners ───
    form.querySelectorAll('.tf-role-option').forEach(function (label) {
        const radio = label.querySelector('input[type="radio"]');
        const sync = function () { label.classList.toggle('is-selected', radio.checked); };
        radio.addEventListener('change', function () {
            form.querySelectorAll('.tf-role-option').forEach(function (o) {
                o.classList.remove('is-selected');
            });
            sync();
            updateSections();
        });
        sync();
    });

    form.querySelectorAll('input[name="empresa_mode"]').forEach(function (r) {
        r.addEventListener('change', updateSections);
    });

    form.addEventListener('input', updateChecklist);
    form.addEventListener('change', updateChecklist);

    // Init diferido para asegurar que el SCRIPT 1 ya está cargado
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', updateSections);
    } else {
        updateSections();
    }
    setTimeout(updateSections, 100);
})();
</script>

<!-- JS: progreso dinámico basado en campos llenos -->
<script>
    (function () {
        const form = document.querySelector('.tf-form-page');
        if (!form) return;

        const blocks = Array.from(form.querySelectorAll('[data-block]'))
            .map(b => b.getAttribute('data-block'));
        if (!blocks.length) return;

        const stepEls = {};
        blocks.forEach(id => {
            stepEls[id] = document.querySelector(`.tf-progress__step[data-step="${id}"]`);
        });

        function updateProgress() {
            blocks.forEach(id => {
                const block = document.getElementById(`block-${id}`);
                if (!block) return;

                const inputs = block.querySelectorAll(
                    'input:not([type="hidden"]):not([type="password"]), select, textarea'
                );
                const total = inputs.length;
                const filled = Array.from(inputs).filter(el => el.value.trim() !== '').length;

                // "Completo" si al menos el 70% de los campos editables tiene valor
                const complete = total === 0 || (filled / total) >= 0.7;

                const step = stepEls[id];
                if (!step) return;
                step.classList.toggle('is-complete', complete);
            });

            // Marca el primer bloque incompleto como "current"
            const firstIncomplete = blocks.find(
                id => !stepEls[id]?.classList.contains('is-complete')
            );
            blocks.forEach(id => stepEls[id]?.classList.remove('is-current'));
            if (firstIncomplete) stepEls[firstIncomplete].classList.add('is-current');
        }

        form.addEventListener('input', updateProgress);
        updateProgress();
    })();
</script>

<script>
(function () {
  const form = document.querySelector('.tf-form-page');
  if (!form) return;

  /* ─────── Mostrar/ocultar contraseña ─────── */
  form.querySelectorAll('[data-toggle-password]').forEach(btn => {
    btn.addEventListener('click', () => {
      const target = document.getElementById(btn.dataset.togglePassword);
      if (!target) return;
      const showing = target.type === 'text';
      target.type = showing ? 'password' : 'text';
      const icon = btn.querySelector('i');
      if (icon) icon.className = showing ? 'bi bi-eye' : 'bi bi-eye-slash';
    });
  });

  /* ─────── Medidor de contraseña ─────── */
  const passwordInput = document.getElementById('password');
  const meter = document.querySelector('[data-meter-for="password"]');
  const hint = document.querySelector('[data-password-hint="password"]');

  const scorePassword = (v) => {
    let s = 0;
    if (v.length >= 8) s++;
    if (/[A-Z]/.test(v) && /[a-z]/.test(v)) s++;
    if (/\d/.test(v)) s++;
    if (/[^A-Za-z0-9]/.test(v)) s++;
    return s;
  };

  const labels = {
    0: 'El usuario deberá cambiarla en su primer inicio de sesión.',
    1: 'Contraseña débil · añade mayúsculas, números y símbolos.',
    2: 'Contraseña aceptable · puedes reforzarla más.',
    3: 'Contraseña fuerte · cumple con los requisitos.',
    4: 'Contraseña excelente · lista para usar.'
  };

  if (passwordInput && meter) {
    passwordInput.addEventListener('input', () => {
      const v = passwordInput.value;
      const s = scorePassword(v);
      meter.dataset.score = String(s);
      meter.querySelectorAll('.tf-password-meter__bar').forEach((bar, i) => {
        bar.classList.toggle('is-active', i < s);
      });
      if (hint) hint.textContent = labels[s];
    });
  }

  /* ─────── Preview en vivo ─────── */
  const preview = {
    name:   document.querySelector('[data-preview="name"]'),
    email:  document.querySelector('[data-preview="email"]'),
    role:   document.querySelector('[data-preview="role"]'),
    avatar: document.querySelector('[data-preview="avatar"]')
  };

  const nameInput  = document.getElementById('name');
  const emailInput = document.getElementById('email');
  const roleInputs = form.querySelectorAll('input[name="role"]');

  const roleIcons = {
    broker:  'bi-briefcase-fill',
    carrier: 'bi-truck'
  };

  function updatePreview() {
    preview.name.textContent  = nameInput?.value.trim()  || 'Sin nombre';
    preview.email.textContent = emailInput?.value.trim() || 'sin-email@—';

    const role = form.querySelector('input[name="role"]:checked')?.value || '';
    preview.role.textContent = role ? role.charAt(0).toUpperCase() + role.slice(1) : '—';
    preview.role.dataset.role = role;

    if (preview.avatar) {
      const ic = preview.avatar.querySelector('i');
      if (ic) ic.className = 'bi ' + (roleIcons[role] || 'bi-person');
    }

    // Permisos del rol
    document.querySelectorAll('[data-role-info]').forEach(el => {
      el.hidden = el.dataset.roleInfo !== role;
    });
  }

  [nameInput, emailInput, ...roleInputs].forEach(el => {
    el?.addEventListener('input', updatePreview);
    el?.addEventListener('change', updatePreview);
  });

  updatePreview();

  /* ─────── Estado seleccionado del role-card ─────── */
  form.querySelectorAll('.tf-role-card').forEach(card => {
    const radio = card.querySelector('input[type="radio"]');
    const sync = () => card.classList.toggle('is-selected', radio.checked);
    radio.addEventListener('change', () => {
      form.querySelectorAll('.tf-role-card').forEach(c => c.classList.remove('is-selected'));
      sync();
    });
    sync();
  });
})();
</script>

<script>
(function () {
    const form = document.getElementById('empresaForm');
    if (!form) return;

    const blocks = ['identity', 'contact', 'location', 'credit'];
    const stepEls = {};
    blocks.forEach(function (id) {
        stepEls[id] = document.querySelector('.tf-progress__step[data-step="' + id + '"]');
    });

    function updateProgress() {
        blocks.forEach(function (id) {
            const block = document.getElementById('block-' + id);
            if (!block) return;
            const required = block.querySelectorAll('[required]');
            let filled = 0;
            required.forEach(function (el) { if (el.value.trim() !== '') filled++; });
            const complete = required.length === 0 || filled === required.length;
            if (stepEls[id]) stepEls[id].classList.toggle('is-complete', complete);
        });

        const firstIncomplete = blocks.find(function (id) {
            return stepEls[id] && !stepEls[id].classList.contains('is-complete');
        });
        blocks.forEach(function (id) { if (stepEls[id]) stepEls[id].classList.remove('is-current'); });
        if (firstIncomplete && stepEls[firstIncomplete]) {
            stepEls[firstIncomplete].classList.add('is-current');
        }
    }

    form.addEventListener('input', updateProgress);
    form.addEventListener('change', updateProgress);
    updateProgress();
})();
</script>

    <script>
        (function () {
            const form = document.querySelector('.tf-form-page');
            if (!form) return;

            const blocks = Array.from(form.querySelectorAll('[data-block]'))
                .map(b => b.getAttribute('data-block'));
            if (!blocks.length) return;

            const stepEls = {};
            blocks.forEach(id => {
                stepEls[id] = form.parentElement.querySelector(`.tf-progress__step[data-step="${id}"]`);
            });

            function updateProgress() {
                blocks.forEach(id => {
                    const block = document.getElementById(`block-${id}`);
                    if (!block) return;

                    const inputs = block.querySelectorAll('input:not([type="hidden"]):not([type="password"]), select, textarea');
                    const total = inputs.length;
                    const filled = Array.from(inputs).filter(el => el.value.trim() !== '').length;

                    // Consideramos "completo" si al menos el 70% de los campos tiene datos
                    const complete = total === 0 || (filled / total) >= 0.7;

                    const step = stepEls[id];
                    if (!step) return;
                    step.classList.toggle('is-complete', complete);
                });

                // Marca el primero incompleto como current
                const firstIncomplete = blocks.find(id => !stepEls[id]?.classList.contains('is-complete'));
                blocks.forEach(id => stepEls[id]?.classList.remove('is-current'));
                if (firstIncomplete) stepEls[firstIncomplete].classList.add('is-current');
            }

            form.addEventListener('input', updateProgress);
            updateProgress();
        })();
    </script>

(function () {
    document.addEventListener('click', function (event) {
        var link = event.target.closest('.tf-pagination a');
        if (!link) return;
        var section = link.closest('#direcciones-results');
        if (!section) return;
        event.preventDefault();
        section.classList.add('is-loading');
        var target = new URL(link.href, window.location.href);
        fetch(target.href, { headers: { 'X-Requested-With': 'XMLHttpRequest' } })
            .then(function (response) { if (!response.ok) throw new Error('No se pudo cargar la página'); return response.text(); })
            .then(function (html) {
                var doc = new DOMParser().parseFromString(html, 'text/html');
                var next = doc.getElementById('direcciones-results');
                if (!next) throw new Error('Respuesta de paginación inválida');
                section.replaceWith(next);
                window.history.pushState({}, '', target.href);
            })
            .catch(function () { window.location.href = target.href; });
    });
})();
</script>

<!-- ═══════════════════════════════════════════════════════════════════

<script>
(function () {
    const NOMINATIM_SEARCH  = 'https://nominatim.openstreetmap.org/search';
    const NOMINATIM_REVERSE = 'https://nominatim.openstreetmap.org/reverse';
    const COUNTRY_CODES     = 'cu';

    const CUBA_CENTER = [21.5218, -77.7812];
    const CUBA_ZOOM   = 7;

    const root = document.querySelector('[data-map-picker]');
    if (!root || typeof L === 'undefined') return;

    const searchInput  = root.querySelector('.tf-map-picker__search-input');
    const resultsBox   = root.querySelector('.tf-map-picker__results');
    const mapEl        = root.querySelector('[data-map]');
    const coordsEl     = root.querySelector('[data-coords]');
    const hintEl       = root.querySelector('[data-address-hint]');

    const latInput     = document.getElementById('latitud');
    const lngInput     = document.getElementById('longitud');
    const cityInput    = document.getElementById('ciudad');
    const stateInput   = document.getElementById('estado_provincia');
    const countryInput = document.getElementById('pais');
    const postalInput  = document.getElementById('codigo_postal');
    const calleInput   = document.getElementById('calle');

    const map = L.map(mapEl, {
        center: CUBA_CENTER,
        zoom: CUBA_ZOOM,
        scrollWheelZoom: true,
        zoomControl: true,
    });

    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
        attribution: '&copy; OpenStreetMap',
        maxZoom: 18,
    }).addTo(map);

    let marker = null;

    function makeIcon() {
        return L.divIcon({
            className: '',
            html: '<div class="tf-map-pin tf-map-pin--pickup"><i class="bi bi-geo-alt-fill"></i></div>',
            iconSize: [32, 32],
            iconAnchor: [16, 32],
        });
    }

    function setMarker(lat, lng, opts) {
        opts = opts || {};
        if (marker) {
            marker.setLatLng([lat, lng]);
        } else {
            marker = L.marker([lat, lng], { icon: makeIcon(), draggable: true }).addTo(map);
            marker.on('dragend', function () {
                const pos = marker.getLatLng();
                setMarker(pos.lat, pos.lng);
            });
        }

        if (opts.pan !== false) map.panTo([lat, lng], { animate: true });

        if (latInput) latInput.value = lat.toFixed(8);
        if (lngInput) lngInput.value = lng.toFixed(8);

        if (coordsEl) {
            coordsEl.dataset.empty = 'false';
            coordsEl.innerHTML = '<i class="bi bi-crosshair"></i> ' + lat.toFixed(5) + ', ' + lng.toFixed(5);
        }

        if (!opts.skipReverse) reverseGeocode(lat, lng);
    }

    let reverseTimer = null;
    function reverseGeocode(lat, lng) {
        clearTimeout(reverseTimer);
        if (hintEl) hintEl.textContent = 'Obteniendo dirección…';

        reverseTimer = setTimeout(function () {
            const url = NOMINATIM_REVERSE + '?format=json&lat=' + lat + '&lon=' + lng + '&zoom=18&addressdetails=1&accept-language=es';
            fetch(url, { headers: { Accept: 'application/json' } })
                .then(function (r) { return r.json(); })
                .then(function (data) {
                    const a = data.address || {};

                    if (calleInput && !calleInput.value.trim()) {
                        calleInput.value = a.road
                            ? (a.road + (a.house_number ? ' #' + a.house_number : ''))
                            : (a.neighbourhood || a.suburb || '');
                    }
                    if (cityInput && !cityInput.value.trim()) {
                        cityInput.value = a.city || a.town || a.village || a.municipality || '';
                    }
                    if (stateInput && !stateInput.value.trim()) {
                        stateInput.value = a.state || a.province || a.region || '';
                    }
                    if (countryInput && (!countryInput.value.trim() || countryInput.value === 'US')) {
                        countryInput.value = (a.country_code || '').toUpperCase() || 'CU';
                    }
                    if (postalInput && !postalInput.value.trim()) {
                        postalInput.value = a.postcode || '';
                    }

                    if (hintEl) hintEl.textContent = data.display_name || 'Ubicación seleccionada';
                })
                .catch(function () {
                    if (hintEl) hintEl.textContent = 'No se pudo obtener la dirección';
                });
        }, 400);
    }

    let searchTimer = null;
    if (searchInput) {
        searchInput.addEventListener('input', function () {
            clearTimeout(searchTimer);
            const q = searchInput.value.trim();
            if (q.length < 3) { resultsBox.hidden = true; return; }
            searchTimer = setTimeout(function () { runSearch(q); }, 450);
        });
    }

    function runSearch(q) {
        const url = NOMINATIM_SEARCH + '?format=json&q=' + encodeURIComponent(q)
                  + '&countrycodes=' + COUNTRY_CODES + '&limit=6&addressdetails=1&accept-language=es';
        fetch(url, { headers: { Accept: 'application/json' } })
            .then(function (r) { return r.json(); })
            .then(renderResults)
            .catch(function () {
                resultsBox.innerHTML = '<div class="tf-map-picker__empty">Error de búsqueda</div>';
                resultsBox.hidden = false;
            });
    }

    function renderResults(items) {
        if (!items || !items.length) {
            resultsBox.innerHTML = '<div class="tf-map-picker__empty">Sin resultados</div>';
            resultsBox.hidden = false;
            return;
        }
        resultsBox.innerHTML = items.map(function (it, idx) {
            return '<button type="button" class="tf-map-picker__result" data-idx="' + idx + '">'
                 + '<i class="bi bi-geo-alt-fill"></i>'
                 + '<span>' + escapeHtml(it.display_name)
                 + '<small>' + escapeHtml((it.type || '')) + '</small></span></button>';
        }).join('');
        resultsBox.hidden = false;
        resultsBox._items = items;

        resultsBox.querySelectorAll('.tf-map-picker__result').forEach(function (btn) {
            btn.addEventListener('click', function () {
                const it = resultsBox._items[Number(btn.dataset.idx)];
                if (!it) return;
                const lat = parseFloat(it.lat);
                const lng = parseFloat(it.lon);
                map.setView([lat, lng], 15);
                setMarker(lat, lng, { skipReverse: true, pan: false });
                if (hintEl) hintEl.textContent = it.display_name;

                const a = it.address || {};
                if (calleInput && !calleInput.value.trim() && (a.road || a.neighbourhood || a.suburb)) {
                    calleInput.value = a.road
                        ? a.road + (a.house_number ? ' #' + a.house_number : '')
                        : (a.neighbourhood || a.suburb);
                }
                if (cityInput && !cityInput.value.trim()) {
                    cityInput.value = a.city || a.town || a.village || a.municipality || '';
                }
                if (stateInput && !stateInput.value.trim()) {
                    stateInput.value = a.state || a.province || a.region || '';
                }
                if (countryInput && (!countryInput.value.trim() || countryInput.value === 'US')) {
                    countryInput.value = (a.country_code || '').toUpperCase() || 'CU';
                }
                if (postalInput && !postalInput.value.trim()) {
                    postalInput.value = a.postcode || '';
                }
                searchInput.value = it.display_name;
                resultsBox.hidden = true;
            });
        });
    }

    document.addEventListener('click', function (e) {
        if (!root.contains(e.target)) resultsBox.hidden = true;
    });

    [latInput, lngInput].forEach(function (inp) {
        if (!inp) return;
        inp.addEventListener('change', function () {
            const lat = parseFloat(latInput && latInput.value);
            const lng = parseFloat(lngInput && lngInput.value);
            if (!isNaN(lat) && !isNaN(lng) && lat !== 0 && lng !== 0) {
                setMarker(lat, lng);
            }

<script>
(function () {
    const NOMINATIM_SEARCH  = 'https://nominatim.openstreetmap.org/search';
    const NOMINATIM_REVERSE = 'https://nominatim.openstreetmap.org/reverse';
    const COUNTRY_CODES     = 'cu';

    const CUBA_CENTER = [21.5218, -77.7812];
    const CUBA_ZOOM   = 7;

    const root = document.querySelector('[data-map-picker]');
    if (!root || typeof L === 'undefined') return;

    const searchInput  = root.querySelector('.tf-map-picker__search-input');
    const resultsBox   = root.querySelector('.tf-map-picker__results');
    const mapEl        = root.querySelector('[data-map]');
    const coordsEl     = root.querySelector('[data-coords]');
    const hintEl       = root.querySelector('[data-address-hint]');

    const latInput     = document.getElementById('latitud');
    const lngInput     = document.getElementById('longitud');
    const cityInput    = document.getElementById('ciudad');
    const stateInput   = document.getElementById('estado_provincia');
    const countryInput = document.getElementById('pais');
    const postalInput  = document.getElementById('codigo_postal');
    const calleInput   = document.getElementById('calle');

    const map = L.map(mapEl, {
        center: CUBA_CENTER,
        zoom: CUBA_ZOOM,
        scrollWheelZoom: true,
        zoomControl: true,
    });

    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
        attribution: '&copy; OpenStreetMap',
        maxZoom: 18,
    }).addTo(map);

    let marker = null;

    function makeIcon() {
        return L.divIcon({
            className: '',
            html: '<div class="tf-map-pin tf-map-pin--pickup"><i class="bi bi-geo-alt-fill"></i></div>',
            iconSize: [32, 32],
            iconAnchor: [16, 32],
        });
    }

    function setMarker(lat, lng, opts) {
        opts = opts || {};
        if (marker) {
            marker.setLatLng([lat, lng]);
        } else {
            marker = L.marker([lat, lng], { icon: makeIcon(), draggable: true }).addTo(map);
            marker.on('dragend', function () {
                const pos = marker.getLatLng();
                setMarker(pos.lat, pos.lng);
            });
        }

        if (opts.pan !== false) map.panTo([lat, lng], { animate: true });

        if (latInput) latInput.value = lat.toFixed(8);
        if (lngInput) lngInput.value = lng.toFixed(8);

        if (coordsEl) {
            coordsEl.dataset.empty = 'false';
            coordsEl.innerHTML = '<i class="bi bi-crosshair"></i> ' + lat.toFixed(5) + ', ' + lng.toFixed(5);
        }

        if (!opts.skipReverse) reverseGeocode(lat, lng);
    }

    let reverseTimer = null;
    function reverseGeocode(lat, lng) {
        clearTimeout(reverseTimer);
        if (hintEl) hintEl.textContent = 'Obteniendo dirección…';

        reverseTimer = setTimeout(function () {
            const url = NOMINATIM_REVERSE + '?format=json&lat=' + lat + '&lon=' + lng + '&zoom=18&addressdetails=1&accept-language=es';
            fetch(url, { headers: { Accept: 'application/json' } })
                .then(function (r) { return r.json(); })
                .then(function (data) {
                    const a = data.address || {};

                    if (calleInput && !calleInput.value.trim()) {
                        calleInput.value = a.road
                            ? (a.road + (a.house_number ? ' #' + a.house_number : ''))
                            : (a.neighbourhood || a.suburb || '');
                    }
                    if (cityInput && !cityInput.value.trim()) {
                        cityInput.value = a.city || a.town || a.village || a.municipality || '';
                    }
                    if (stateInput && !stateInput.value.trim()) {
                        stateInput.value = a.state || a.province || a.region || '';
                    }
                    if (countryInput && (!countryInput.value.trim() || countryInput.value === 'US')) {
                        countryInput.value = (a.country_code || '').toUpperCase() || 'CU';
                    }
                    if (postalInput && !postalInput.value.trim()) {
                        postalInput.value = a.postcode || '';
                    }

                    if (hintEl) hintEl.textContent = data.display_name || 'Ubicación seleccionada';
                })
                .catch(function () {
                    if (hintEl) hintEl.textContent = 'No se pudo obtener la dirección';
                });
        }, 400);
    }

    let searchTimer = null;
    if (searchInput) {
        searchInput.addEventListener('input', function () {
            clearTimeout(searchTimer);
            const q = searchInput.value.trim();
            if (q.length < 3) { resultsBox.hidden = true; return; }
            searchTimer = setTimeout(function () { runSearch(q); }, 450);
        });
    }

    function runSearch(q) {
        const url = NOMINATIM_SEARCH + '?format=json&q=' + encodeURIComponent(q)
                  + '&countrycodes=' + COUNTRY_CODES + '&limit=6&addressdetails=1&accept-language=es';
        fetch(url, { headers: { Accept: 'application/json' } })
            .then(function (r) { return r.json(); })
            .then(renderResults)
            .catch(function () {
                resultsBox.innerHTML = '<div class="tf-map-picker__empty">Error de búsqueda</div>';
                resultsBox.hidden = false;
            });
    }

    function renderResults(items) {
        if (!items || !items.length) {
            resultsBox.innerHTML = '<div class="tf-map-picker__empty">Sin resultados</div>';
            resultsBox.hidden = false;
            return;
        }
        resultsBox.innerHTML = items.map(function (it, idx) {
            return '<button type="button" class="tf-map-picker__result" data-idx="' + idx + '">'
                 + '<i class="bi bi-geo-alt-fill"></i>'
                 + '<span>' + escapeHtml(it.display_name)
                 + '<small>' + escapeHtml((it.type || '')) + '</small></span></button>';
        }).join('');
        resultsBox.hidden = false;
        resultsBox._items = items;

        resultsBox.querySelectorAll('.tf-map-picker__result').forEach(function (btn) {
            btn.addEventListener('click', function () {
                const it = resultsBox._items[Number(btn.dataset.idx)];
                if (!it) return;
                const lat = parseFloat(it.lat);
                const lng = parseFloat(it.lon);
                map.setView([lat, lng], 15);
                setMarker(lat, lng, { skipReverse: true, pan: false });
                if (hintEl) hintEl.textContent = it.display_name;

                const a = it.address || {};
                if (calleInput && !calleInput.value.trim() && (a.road || a.neighbourhood || a.suburb)) {
                    calleInput.value = a.road
                        ? a.road + (a.house_number ? ' #' + a.house_number : '')
                        : (a.neighbourhood || a.suburb);
                }
                if (cityInput && !cityInput.value.trim()) {
                    cityInput.value = a.city || a.town || a.village || a.municipality || '';
                }
                if (stateInput && !stateInput.value.trim()) {
                    stateInput.value = a.state || a.province || a.region || '';
                }
                if (countryInput && (!countryInput.value.trim() || countryInput.value === 'US')) {
                    countryInput.value = (a.country_code || '').toUpperCase() || 'CU';
                }
                if (postalInput && !postalInput.value.trim()) {
                    postalInput.value = a.postcode || '';
                }
                searchInput.value = it.display_name;
                resultsBox.hidden = true;
            });
        });
    }

    document.addEventListener('click', function (e) {
        if (!root.contains(e.target)) resultsBox.hidden = true;
    });

    [latInput, lngInput].forEach(function (inp) {
        if (!inp) return;
        inp.addEventListener('change', function () {
            const lat = parseFloat(latInput && latInput.value);
            const lng = parseFloat(lngInput && lngInput.value);
            if (!isNaN(lat) && !isNaN(lng) && lat !== 0 && lng !== 0) {
                setMarker(lat, lng);
            }
        });
    });

    const initLat = parseFloat(root.dataset.initialLat);
    const initLng = parseFloat(root.dataset.initialLng);
    if (!isNaN(initLat) && !isNaN(initLng) && initLat !== 0 && initLng !== 0) {
        map.setView([initLat, initLng], 14);
        setMarker(initLat, initLng, { skipReverse: true, pan: false });
    }

    map.on('click', function (e) {
        setMarker(e.latlng.lat, e.latlng.lng);
    });

    setTimeout(function () { map.invalidateSize(); }, 200);
    window.addEventListener('resize', function () { map.invalidateSize(); });

    function escapeHtml(s) {
        return String(s)
            .replace(/&/g, '&amp;').replace(/</g, '&lt;')
            .replace(/>/g, '&gt;').replace(/"/g, '&quot;');
    }
})();

    var candidates = links.filter(function (a) {
